drop database project ;
create database project ;
use project ; 
 
 create table company (
 company_ID int primary key 
 );
 
create table Reposts (
   Report_ID int primary key not null,
   start_Date date,
   end_date date,
   total_sales int,
   company_ID int,
   foreign key (company_ID) references company(company_ID) on UPDATE CASCADE  ON DELETE CASCADE
);

 
 create table Branch (
 Branch_ID int primary key not null, 
 Address varchar(32),
 company_ID int ,
 FOREIGN KEY (company_ID) REFERENCES Company(company_ID)ON UPDATE CASCADE ON DELETE CASCADE
 );
 
  create table Employea (
 Employea_ID int primary key not null , 
 sname varchar (32) , 
 Address varchar(32) ,
 phone_Number int , 
 sRole varchar(32),
 Salary int,
 Branch_ID int ,
 foreign KEY (Branch_ID) references Branch(Branch_ID)ON UPDATE CASCADE ON DELETE CASCADE

 );
 create table Payments(
 Payments_ID int primary key not null , 
 adate date , 
 Amount int , 
 method varchar(32) ,
 transaction_ID int ,
 Branch_ID int , 
 foreign KEY (Branch_ID) references Branch(Branch_ID)ON UPDATE CASCADE ON DELETE CASCADE
 );
  create table Customer (
  Customer_id INT AUTO_INCREMENT PRIMARY KEY , 
  sname varchar(32) , 
  phone_Number int , 
  address varchar(32)
  
 );
 
 create table orders (
  orders_id int AUTO_INCREMENT primary key, 
  sname varchar(32) , 
  phone_Number int , 
  address varchar(32),
  Customer_id int ,
foreign key(Customer_id) references Customer(Customer_id) ON UPDATE CASCADE ON DELETE CASCADE

 );
 create table care_inventory (
 car_ID int primary key not null , 
 price int , 
 model_year date , 
 fuel_type varchar(32) ,
 color varchar(32) ,
 motor_size int , 
 mator_infarmation varchar(32) ,
 modal varchar(32) ,
 make varchar(32) ,
 vin int ,
 Branch_ID int ,
 orders_id int , 
  foreign KEY (orders_id) references orders(orders_id)ON UPDATE CASCADE ON DELETE CASCADE
,
 foreign KEY (Branch_ID) references Branch(Branch_ID)ON UPDATE CASCADE ON DELETE CASCADE

 );
 CREATE TABLE order_car (
     order_car int AUTO_INCREMENT primary key ,
    orders_id INT NOT NULL,
    car_ID INT NOT NULL,
    FOREIGN KEY (orders_id) REFERENCES orders(orders_id) ON DELETE CASCADE ON UPDATE CASCADE,
    FOREIGN KEY (car_ID) REFERENCES care_inventory(car_ID) ON DELETE CASCADE ON UPDATE CASCADE
);

 create table supplier (
 supplier_id int primary key not null, 
 sname varchar (32) ,
 phone_Number int , 
 Address varchar(32) , 
 Rating varchar(32),
 car_ID int , 
 foreign KEY (car_ID) references care_inventory(car_ID)ON UPDATE CASCADE ON DELETE CASCADE
 );


 
CREATE TABLE users (
    user_id INT AUTO_INCREMENT PRIMARY KEY,
    email VARCHAR(255) UNIQUE NOT NULL,
    password1 VARCHAR(255) NOT NULL,
    sname VARCHAR(50) NOT NULL,
    role ENUM('admin', 'employee', 'customer') NOT NULL DEFAULT 'customer'
);

 create table transaction1(
  transaction_ID int primary key not null, 
  car_ID int , 
  customar_ID int , 
  supplier_ID int , 
  transaction_Type varchar(32) ,
  Amount int ,
  company_ID int ,
  foreign key(company_ID) references company(company_ID) ON UPDATE CASCADE ON DELETE CASCADE,
  foreign KEY (car_ID) references care_inventory(car_ID) ON UPDATE CASCADE ON DELETE CASCADE,
  foreign KEY (supplier_ID) references Supplier(supplier_ID)ON UPDATE CASCADE ON DELETE CASCADE
 );
 
 create table Customer_Branch  (
 Customer_id int not null, 
 Branch_ID int not null, 
 foreign KEY (Customer_id) references Customer(Customer_id) ON UPDATE CASCADE ON DELETE CASCADE,
 foreign KEY (Branch_ID) references Branch(Branch_ID) ON UPDATE CASCADE ON DELETE CASCADE

);
